Paste your key into the snippet below, put it in a LocalScript, and call askSammy("your question").

-- Sammy Bot client — ask questions about SpyderSammy from in-game local HttpService = game:GetService("HttpService") local API_URL = "WORKER_URL_HERE/chat" local API_KEY = "YOUR_KEY_HERE" local function askSammy(question) local body = HttpService:JSONEncode({ key = API_KEY, message = question }) local ok, result = pcall(function() return HttpService:PostAsync(API_URL, body, Enum.HttpContentType.ApplicationJson) end) if not ok then warn("Sammy Bot request failed: " .. tostring(result)) return nil end local decoded = HttpService:JSONDecode(result) if decoded.error then warn("Sammy Bot error: " .. decoded.error) return nil end return decoded.reply end -- Example usage: local reply = askSammy("When did Steal a Brainrot come out?") print(reply)
